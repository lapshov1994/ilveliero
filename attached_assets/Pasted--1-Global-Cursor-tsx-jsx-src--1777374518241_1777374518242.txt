Шаг 1: Кастомный курсор (Global)

Создай файл Cursor.tsx (или .jsx) в папке src. Это будет глобальный компонент.

TypeScript
import React, { useEffect, useRef, useState } from 'react';
import gsap from 'gsap';

export default function Cursor() {
  const cursorRef = useRef(null);
  const followerRef = useRef(null);
  const [isHovering, setIsHovering] = useState(false);

  useEffect(() => {
    // 1. Анимация движения курсора и фолловера
    const onMouseMove = (e: MouseEvent) => {
      // Маленькая точка (точный курсор)
      gsap.to(cursorRef.current, {
        x: e.clientX,
        y: e.clientY,
        duration: 0.1,
        ease: "power2.out"
      });
      
      // Большой кружок (фолловер - более плавный)
      gsap.to(followerRef.current, {
        x: e.clientX,
        y: e.clientY,
        duration: 0.4,
        ease: "power2.out"
      });
    };

    // 2. Реакция на ссылки и кнопки
    const onMouseEnter = () => setIsHovering(true);
    const onMouseLeave = () => setIsHovering(false);

    window.addEventListener('mousemove', onMouseMove);

    // Ищем все интерактивные элементы
    const links = document.querySelectorAll('a, button, [role="button"]');
    links.forEach(link => {
      link.addEventListener('mouseenter', onMouseEnter);
      link.addEventListener('mouseleave', onMouseLeave);
    });

    return () => {
      window.removeEventListener('mousemove', onMouseMove);
      links.forEach(link => {
        link.removeEventListener('mouseenter', onMouseEnter);
        link.removeEventListener('mouseleave', onMouseLeave);
      });
    };
  }, []);

  useEffect(() => {
    // Анимация при наведении: фолловер расширяется, точка исчезает
    if (isHovering) {
      gsap.to(followerRef.current, { scale: 2, borderColor: '#D4AF37', borderOpacity: 0.2, duration: 0.3 });
      gsap.to(cursorRef.current, { opacity: 0, duration: 0.2 });
    } else {
      gsap.to(followerRef.current, { scale: 1, borderColor: '#D4AF37', borderOpacity: 1, duration: 0.3 });
      gsap.to(cursorRef.current, { opacity: 1, duration: 0.2 });
    }
  }, [isHovering]);

  return (
    <>
      {/* Только на десктопе */}
      <div className="hidden md:block pointer-events-none fixed inset-0 z-[9999]">
        {/* Точка курсора */}
        <div 
          ref={cursorRef} 
          className="fixed top-0 left-0 w-1 h-1 bg-[#D4AF37] -translate-x-1/2 -translate-y-1/2"
        />
        {/* Фолловер */}
        <div 
          ref={followerRef} 
          className="fixed top-0 left-0 w-8 h-8 border border-[#D4AF37] rounded-full -translate-x-1/2 -translate-y-1/2"
        />
      </div>
    </>
  );
}
Технический нюанс: убедись, что в tailwind.config.js у тебя нет конфликтов с pointer-events-none.

Замени код в App.tsx на этот:

TypeScript
import React, { useEffect } from 'react';
import Hero from './Hero'; 
import Story from './Story'; 
import Family from './Family'; 
import Footer from './Footer'; 
import Cursor from './Cursor'; // <--- Импортируем курсор
import Lenis from '@studio-freight/lenis';
import gsap from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';

gsap.registerPlugin(ScrollTrigger);

export default function App() {
  useEffect(() => {
    // Настройки Lenis ... (оставляем как было)
    const lenis = new Lenis({
      duration: 1.2,
      easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
      smooth: true,
    });

    lenis.on('scroll', ScrollTrigger.update);
    gsap.ticker.add((time) => {
      lenis.raf(time * 1000);
    });
    gsap.ticker.lagSmoothing(0);

    return () => {
      gsap.ticker.remove(lenis.raf);
      lenis.destroy();
    };
  }, []);

  return (
    <main className="bg-white min-h-screen relative overflow-hidden font-sans cursor-none md:cursor-auto"> {/* cursor-none чтобы не было дефолтного на десктопе */}
      
      {/* --- ДЕТАЛЬ 4: Кастомный курсор --- */}
      <Cursor />

      {/* --- ДЕТАЛЬ 5: Текстура Зерна (Artigianato) --- */}
      {/* Накладываем поверх всего сайта прозрачный статичный шум */}
      <div className="pointer-events-none fixed inset-0 z-[100] opacity-[0.02] bg-[url('data:image/png;base64,iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAYAAAAfFcSJAAAADUlEQVR42mP8z8BQDwAEhQGAhKmMIQAAAABJRU5ErkJggg==')] bg-repeat"></div>

      <Hero />
      <Story />
      <Family />
      <Footer />
    </main>
  );
}
